hma(source, length) =>
    half  = math.max(1, math.round(length / 2))
    sqrtl = math.max(1, math.round(math.sqrt(length)))
    ta.wma(2.0 * ta.wma(source, half) - ta.wma(source, length), sqrtl)

dema(source, length) =>
    e = ta.ema(source, length)
    2.0 * e - ta.ema(e, length)

tema(source, length) =>
    e1 = ta.ema(source, length)
    e2 = ta.ema(e1, length)
    e3 = ta.ema(e2, length)
    3.0 * e1 - 3.0 * e2 + e3

// Switch default fallback ke EMA (sebelumnya implicit, sekarang explicit)
getMA(source, length, type) =>
    switch type
        "SMA"  => ta.sma(source, length)
        "EMA"  => ta.ema(source, length)
        "WMA"  => ta.wma(source, length)
        "RMA"  => ta.rma(source, length)
        "HMA"  => hma(source, length)
        "DEMA" => dema(source, length)
        "TEMA" => tema(source, length)
        =>        ta.ema(source, length)

mad(source, length) =>
    ta.sma(math.abs(source - ta.sma(source, length)), length)

kamaCalc(source, length, prev) =>
    fastSC    = 2.0 / 3.0
    slowSC    = 2.0 / 31.0
    direction = math.abs(source - source[length])
    noise     = math.sum(math.abs(source - source[1]), length)
    er        = noise != 0.0 ? direction / noise : 0.0
    sc        = math.pow(er * (fastSC - slowSC) + slowSC, 2)
    na(prev) ? source : prev + sc * (source - prev)
