        if not c.active and close < c.aVal
            if not na(c.bLine)
                line.delete(c.bLine)
            array.remove(bullCands, i)
            continue

        // Aktivasi: breakout level B kandidat + buffer ATR
        justAct = false
        if not c.active and close > c.bVal
            distOk = not enableAtrFilter or (close - c.bVal) >= atrMult * atrVal
            if distOk
                c.active  := true
                c.persist := 1

                    c.aVal := _a3BullValRaw
                    c.aBar := _a3BullBarRaw

                    f_pushLabelFade(bullBreakLblArr, bullBreakLbBarArr, bullBreakLbTimeArr, newLb)
