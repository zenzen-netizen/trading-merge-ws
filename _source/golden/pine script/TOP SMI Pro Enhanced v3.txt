// ===================================================
// SMI CALCULATION
// ===================================================
emaEma(s, l) => ta.ema(ta.ema(s, l), l)

hh     = ta.highest(lenK)
ll     = ta.lowest(lenK)
rel    = close - (hh + ll) / 2
rng    = hh - ll
smi    = 200 * (emaEma(rel, lenD) / emaEma(rng, lenD))
smiEma = ta.ema(smi, lenE)
smiH   = smi - smiEma

sAb    = smi >= smiEma
sHAU   = smiH > smiH[1] and smiH  > 0
sHAD   = smiH < smiH[1] and smiH  > 0
sHBD   = smiH < smiH[1] and smiH <= 0
sHBU   = smiH > smiH[1] and smiH <= 0

sLC    = sAb ? sBull : sBear
sHC    = sHAU ? hAU : sHAD ? hAD : sHBD ? hBD : sHBU ? hBU : color.gray

sCr    = ta.cross(smi, smiEma)
sCU    = ta.crossover(smi, smiEma)
sCD    = ta.crossunder(smi, smiEma)

    bCnt := bCnt + 1
    aCnt := 0
else if smi > smiMID
    aCnt := aCnt + 1
    bCnt := 0
else
    bCnt := 0
    aCnt := 0

preAkum  = bCnt == akumCandles
preDist  = aCnt == distCandles

// ===================================================
// PA/PD + HISTOGRAM MOMENTUM CLASSIFICATION (v3 baru)
// ===================================================

paDip   = preAkum and paMomentumFalling
paReady = preAkum and paMomentumRising

pdDip   = preDist and paMomentumRising
pdReady = preDist and paMomentumFalling

// ===================================================
// STATE DETECTION v3 - FAILED MID HOLD (fakeout warning)
// ===================================================
failedMIDHoldBuy  = (smi > smiMID) and (smi[1] < smiMID) and (bCnt[1] > 0)
failedMIDHoldSell = (smi < smiMID) and (smi[1] > smiMID) and (aCnt[1] > 0)

// ===================================================

    sLB := bar_index

// ===================================================
// DIVERGENCE DETECTION - pivot SMI dibanding pivot harga
// ===================================================
plFound = not na(ta.pivotlow(smi, divLbL, divLbR))
phFound = not na(ta.pivothigh(smi, divLbL, divLbR))

var float lastPlPrice = na
var int   lastPlBar   = na
var float lastPhOsc   = na
var float lastPhPrice = na
var int   lastPhBar   = na

if plFound
    curPlBar   = bar_index - divLbR
    curPlOsc   = smi[divLbR]
    curPlPrice = low[divLbR]
    if not na(lastPlBar)
        barsGap = curPlBar - lastPlBar
        if barsGap >= divRangeMin and barsGap <= divRangeMax
            if showDiv and curPlPrice < lastPlPrice and curPlOsc > lastPlOsc
                line.new(lastPlBar, lastPlOsc, curPlBar, curPlOsc, xloc=xloc.bar_index, color=divBullCol, width=2)
                label.new(curPlBar, curPlOsc, text="Bull", color=color.new(color.black, 100), textcolor=divBullCol, style=label.style_label_up, size=size.small, xloc=xloc.bar_index)
            if showHiddenDiv and curPlPrice > lastPlPrice and curPlOsc < lastPlOsc
                line.new(lastPlBar, lastPlOsc, curPlBar, curPlOsc, xloc=xloc.bar_index, color=divHiddenBullCol, width=1, style=line.style_dashed)
                label.new(curPlBar, curPlOsc, text="H.Bull", color=color.new(color.black, 100), textcolor=divHiddenBullCol, style=label.style_label_up, size=size.small, xloc=xloc.bar_index)
    lastPlOsc   := curPlOsc
    lastPlPrice := curPlPrice
    lastPlBar   := curPlBar

if phFound
    curPhBar   = bar_index - divLbR
    curPhOsc   = smi[divLbR]
    curPhPrice = high[divLbR]
    if not na(lastPhBar)
        barsGap = curPhBar - lastPhBar
        if barsGap >= divRangeMin and barsGap <= divRangeMax
            if showDiv and curPhPrice > lastPhPrice and curPhOsc < lastPhOsc
                line.new(lastPhBar, lastPhOsc, curPhBar, curPhOsc, xloc=xloc.bar_index, color=divBearCol, width=2)
                label.new(curPhBar, curPhOsc, text="Bear", color=color.new(color.black, 100), textcolor=divBearCol, style=label.style_label_down, size=size.small, xloc=xloc.bar_index)
            if showHiddenDiv and curPhPrice < lastPhPrice and curPhOsc > lastPhOsc
                line.new(lastPhBar, lastPhOsc, curPhBar, curPhOsc, xloc=xloc.bar_index, color=divHiddenBearCol, width=1, style=line.style_dashed)
                label.new(curPhBar, curPhOsc, text="H.Bear", color=color.new(color.black, 100), textcolor=divHiddenBearCol, style=label.style_label_down, size=size.small, xloc=xloc.bar_index)
    lastPhOsc   := curPhOsc
    lastPhPrice := curPhPrice
    lastPhBar   := curPhBar
