f_atr(len, smooth) =>
    float tr_ = ta.tr(true)
    float out = switch smooth
        "EMA" => ta.ema(tr_, len)
        "SMA" => ta.sma(tr_, len)
        "WMA" => ta.wma(tr_, len)
        =>       ta.rma(tr_, len)
    out

f_mad(src, len) =>
    float med = ta.median(src, len)
    float dev = math.abs(src - med)
    ta.median(dev, len)

f_sz(s) =>
    switch s
        "Tiny"   => size.tiny
        "Normal" => size.normal
        "Large"  => size.large
        =>          size.small

f_pos(s) =>
    switch s
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        =>               position.top_right

// ═══════════════════════════════════════════════════════════════════
// ROBUST CALCULATION
// ═══════════════════════════════════════════════════════════════════
float MAD_MULT = 2.5

f_robust(array<float> vals, array<float> wgts) =>
    float med  = array.median(vals)
    array<float> devs = array.new_float()
    for i = 0 to array.size(vals) - 1
        array.push(devs, math.abs(array.get(vals, i) - med))
    float mad = array.median(devs)
    float lo  = med - mad * MAD_MULT
    float hi  = med + mad * MAD_MULT
    float num = 0.0
    float den = 0.0
    for i = 0 to array.size(vals) - 1
        float v = math.max(lo, math.min(hi, array.get(vals, i)))
        float w = array.get(wgts, i)
        num += v * w
        den += w
    den > 0 ? num / den : med

    float b2 = math.abs(close - open)

        "Opposite"       => (c1_bull_loc and c2_bear_loc) or (c1_bear_loc and c2_bull_loc)

    bool t2 = shared and crt_t2_on and ((c1_bear_loc and low >= l1) or (c1_bull_loc and high <= h1))

    crt_is_t2 := t2

    crt_bear := (t1 or t2) and c1_bull_loc
